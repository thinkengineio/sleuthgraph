# Phase 1 — Foundation (Repos + Infra)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Scaffold the three repos (meta, api, web), stand up infrastructure (Postgres+AGE, Redis, MinIO) via Docker Compose, and produce a "hello world" deploy where `docker compose up` from the meta repo brings up every service with health checks passing.

**Architecture:** Three Git repos — `sleuthgraph` (meta: docker-compose, docs, deploy glue), `sleuthgraph-api` (FastAPI backend), `sleuthgraph-web` (Next.js frontend). Docker Compose lives in the meta repo and builds the api/web images from sibling repo directories during local dev; in production it pulls tagged images from GHCR.

**Tech Stack:** Python 3.12, FastAPI, uv/pip, SQLAlchemy 2.x, Alembic, Postgres 16 + Apache AGE, Redis 7, MinIO, Next.js 15, TypeScript, pnpm, Tailwind, shadcn/ui, Docker Compose v2, ruff, pytest, eslint, prettier, vitest.

### Task 1.4: Write Makefile

**Files:**
- Create: `~/sleuthgraph/deploy/Makefile`

- [ ] **Step 1: Write Makefile**

Path: `~/sleuthgraph/deploy/Makefile`
```makefile
# Sleuthgraph local dev Makefile.
# Run from ~/sleuthgraph/deploy/.

.PHONY: up down logs ps shell-db shell-redis shell-api shell-web test clean rebuild

ENV_FILE := .env
COMPOSE := docker compose --env-file $(ENV_FILE) -f docker-compose.yml -f docker-compose.dev.yml

up:
	$(COMPOSE) up -d
	@echo "API:  http://localhost:$${API_PORT:-8000}/docs"
	@echo "Web:  http://localhost:$${WEB_PORT:-3000}"
	@echo "MinIO console: http://localhost:9001"

down:
	$(COMPOSE) down

logs:
	$(COMPOSE) logs -f --tail=100

ps:
	$(COMPOSE) ps

shell-db:
	$(COMPOSE) exec db psql -U $${POSTGRES_USER} $${POSTGRES_DB}

shell-redis:
	$(COMPOSE) exec redis redis-cli

shell-api:
	$(COMPOSE) exec api bash

shell-web:
	$(COMPOSE) exec web sh

test:
	$(COMPOSE) exec api pytest
	$(COMPOSE) exec web pnpm test

rebuild:
	$(COMPOSE) build --no-cache
	$(COMPOSE) up -d

clean:
	$(COMPOSE) down -v
	@echo "All containers, volumes, and networks removed."
```

### Task 1.5: Write docker-compose.yml (services only, no api/web yet)

**Files:**
- Create: `~/sleuthgraph/deploy/docker-compose.yml`

- [ ] **Step 1: Write docker-compose.yml**

Path: `~/sleuthgraph/deploy/docker-compose.yml`
```yaml
# Sleuthgraph local dev compose.
# Services: db (postgres+AGE), redis, minio, api, web.
# Dev overrides live in docker-compose.dev.yml (volume mounts, hot reload).

services:
  db:
    image: apache/age:PG16_latest
    restart: unless-stopped
    environment:
      POSTGRES_USER: ${POSTGRES_USER}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_DB: ${POSTGRES_DB}
    volumes:
      - db_data:/var/lib/postgresql/data
      - ./postgres/init-age.sql:/docker-entrypoint-initdb.d/10-init-age.sql:ro
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}"]
      interval: 5s
      timeout: 3s
      retries: 10
    ports:
      - "5432:5432"

  redis:
    image: redis:7-alpine
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 10
    ports:
      - "6379:6379"

  minio:
    image: minio/minio:latest
    restart: unless-stopped
    command: server /data --console-address ":9001"
    environment:
      MINIO_ROOT_USER: ${MINIO_ROOT_USER}
      MINIO_ROOT_PASSWORD: ${MINIO_ROOT_PASSWORD}
    volumes:
      - minio_data:/data
    healthcheck:
      test: ["CMD", "curl", "-fsS", "http://localhost:9000/minio/health/live"]
      interval: 10s
      timeout: 5s
      retries: 5
    ports:
      - "9000:9000"  # S3 API
      - "9001:9001"  # Web console

  # minio-bootstrap creates the evidence bucket on first run.
  minio-bootstrap:
    image: minio/mc:latest
    depends_on:
      minio:
        condition: service_healthy
    environment:
      MINIO_ROOT_USER: ${MINIO_ROOT_USER}
      MINIO_ROOT_PASSWORD: ${MINIO_ROOT_PASSWORD}
      MINIO_BUCKET: ${MINIO_BUCKET}
    entrypoint: >
      /bin/sh -c "
      mc alias set local http://minio:9000 $${MINIO_ROOT_USER} $${MINIO_ROOT_PASSWORD} &&
      (mc mb local/$${MINIO_BUCKET} 2>/dev/null || true) &&
      mc anonymous set none local/$${MINIO_BUCKET}
      "
    restart: "no"

  api:
    image: sleuthgraph-api:dev
    build:
      context: ../../sleuthgraph-api
      dockerfile: Dockerfile
    restart: unless-stopped
    depends_on:
      db:
        condition: service_healthy
      redis:
        condition: service_healthy
      minio:
        condition: service_healthy
    environment:
      DATABASE_URL: ${DATABASE_URL}
      REDIS_URL: ${REDIS_URL}
      S3_ENDPOINT: ${S3_ENDPOINT}
      S3_ACCESS_KEY: ${S3_ACCESS_KEY}
      S3_SECRET_KEY: ${S3_SECRET_KEY}
      S3_BUCKET: ${S3_BUCKET}
      SECRET_KEY: ${SECRET_KEY}
      CORS_ORIGINS: ${CORS_ORIGINS}
      ANTHROPIC_API_KEY: ${ANTHROPIC_API_KEY}
    ports:
      - "${API_PORT:-8000}:8000"
    healthcheck:
      test: ["CMD", "python", "-c", "import urllib.request; urllib.request.urlopen('http://localhost:8000/health')"]
      interval: 10s
      timeout: 5s
      retries: 10
      start_period: 20s

  web:
    image: sleuthgraph-web:dev
    build:
      context: ../../sleuthgraph-web
      dockerfile: Dockerfile
    restart: unless-stopped
    depends_on:
      api:
        condition: service_healthy
    environment:
      NEXT_PUBLIC_API_URL: ${NEXT_PUBLIC_API_URL}
    ports:
      - "${WEB_PORT:-3000}:3000"
    healthcheck:
      test: ["CMD", "wget", "-qO-", "http://localhost:3000"]
      interval: 10s
      timeout: 5s
      retries: 10
      start_period: 30s

volumes:
  db_data:
  minio_data:
```
